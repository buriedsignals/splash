# S4d wiring (I-1) — make the editorial gate live Implementation Plan

> **For agentic workers:** REQUIRED SUB-SKILL: Use superpowers:subagent-driven-development (recommended) or superpowers:executing-plans to implement this plan task-by-task. Steps use checkbox (`- [ ]`) syntax for tracking.

**Goal:** Wire the S4d editorial sign-off mechanism into the live pipeline — an `apply-signoff` command to record an editor's signature, and an `assertEditoriallyCleared` call in the export entrypoints so `requiredSigners` actually blocks export until every required editor has signed the exact shipped artifact.

**Architecture:** A new `apply-signoff.mjs` mirrors the existing `gate-render.mjs` (load report JSON → apply → write back). The two export entrypoints (`export-code.mjs`, `deploy-embed.mjs`) load the newsroom profile (optional `--profile`) and call `assertEditoriallyCleared` against the primary artifact bytes right after `assertShippable`, printing an `EDITORIAL:` line. Opt-in preserved: absent profile / no `requiredSigners` → export proceeds and records `unsigned`.

**Tech Stack:** Bun, TypeScript, `bun:test`, `node:crypto` (existing S4d modules).

Absent → an empty profile (`{ palette: [] }`) → `assertEditoriallyCleared` never throws and returns `{ unsigned: true }` (or `{signedBy}` for any valid optional sign-offs). - Re-verify against the **format's primary artifact bytes** (the owned file being shipped): static → the media image, video → the `.mp4`, interactive/scrolly → the local `.html`; deploy-embed → its staged local artifact.
- **Known documented edge (do NOT try to fix here):** export-code's hosted-DW-embed sub-path (`publicUrl` set, no local html) has no owned bytes to re-hash — the editorial gate is skipped there with a printed note. Record it as a follow-up; do not invent a half-verified variant.

```js
// skills/splash/scripts/apply-signoff.mjs
import { readFileSync, writeFileSync } from "node:fs";
import { applyEditorialSignoff } from "../src/editorial-signoff.ts";
import { parseNewsroomMarkdown } from "../src/brand-profile.ts";

/** Verify + record an editor's signature into a report (pure). Throws on any verification failure. */
export function recordSignoff(report, id, signerId, signature, profileMarkdown) {
  const profile = parseNewsroomMarkdown(profileMarkdown);
  const signers = profile?.signers ?? [];
  return applyEditorialSignoff(report, id, { signerId, signature }, signers);
}

function fail(msg) {
  console.error(msg);
  process.exit(1);
}

function main() {
  const argv = process.argv.slice(2);
  const [reportPath, id] = argv;
  const flag = (name) => {
    const i = argv.indexOf(name);
    return i !== -1 ? argv[i + 1] : undefined;
  };
  const signerId = flag("--signer");
  const signature = flag("--signature");
  const profilePath = flag("--profile");
  if (!reportPath || !id || !signerId || !signature || !profilePath)
    fail("usage: apply-signoff.mjs <report.json> <proposalId> --signer <id> --signature <base64> --profile <NEWSROOM-PROFILE.md>");
  let report, profileMarkdown;
  try { report = JSON.parse(readFileSync(reportPath, "utf8")); } catch { return fail(`cannot read report ${reportPath}`); }
  try { profileMarkdown = readFileSync(profilePath, "utf8"); } catch { return fail(`cannot read profile ${profilePath}`); }
  let next;
  try { next = recordSignoff(report, id, signerId, signature, profileMarkdown); }
  catch (e) { return fail(e.message); }
  writeFileSync(reportPath, JSON.stringify(next, null, 2));
  console.log(`recorded editorial sign-off from ${signerId} for ${id}`);
}

if (import.meta.main) main();
```

```ts
// Build a minimal static-format outDir + report whose approvedHash is the media bytes' hash.
function scaffold(withProfileRequiring) {
  const dir = mkdtempSync(join(tmpdir(), "s4d-exp-"));
  const outDir = join(dir, "out");
  mkdirSync(outDir, { recursive: true });
  const media = Buffer.from("PNGBYTES-static-artifact");
  writeFileSync(join(outDir, "static.png"), media);
  const H = sha256Hex(media);
  let report = {
    results: [{ id: "p1", status: "produced", format: "static", reviewed: true, renderApproved: true, approvedHash: H,
      outDir, outputs: ["static.png"] }],
    generatedAt: new Date(0).toISOString(),

  const { privatePem, signersLine } = generateEditorKeypair("yvan");
  const profileMd = `---\nsigners:\n${signersLine}\n${withProfileRequiring ? "requiredSigners:\n  - yvan\n" : ""}---\n# N`;
  const profilePath = join(dir, "NEWSROOM-PROFILE.md");
  writeFileSync(profilePath, profileMd);
  const reportPath = join(dir, "report.json");
  return { dir, outDir, reportPath, profilePath, report, media, privatePem, H };
}

test("export-code REFUSES a static export when a requiredSigner has not signed", () => {
  const s = scaffold(true);
  writeFileSync(s.reportPath, JSON.stringify(s.report));
  expect(() =>
    execFileSync("bun", [EXPORT_CODE, s.outDir, join(s.dir, "exp"), "--results", s.reportPath, "--id", "p1", "--profile", s.profilePath], { stdio: "pipe" }),
```
